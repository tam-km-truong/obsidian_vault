### calculating number of phrases z
```bash
python experiments/078_EXPE_comp_measure_z_2/scripts/data_preparation.py -n 50 --max 200

cp experiments/078_EXPE_comp_measure_z_2/data/batches/ experiments/078_EXPE_comp_measure_z_2/data/batches_path -rv

for f in experiments/078_EXPE_comp_measure_z_2/data/batches_path/*.txt; do sed -i 's|^|/projects/reall/tam/data/ATB/Uncompressed/| ; s|$|.fa.gz|' $f; done

for f in experiments/078_EXPE_comp_measure_z_2/data/batches_path/*.txt; do cat $f | xargs pigz -d -c > experiments/078_EXPE_comp_measure_z_2/data/batches_concat/$(basename $f .txt).fa; done

cd /scratch/ktruong/tools/PFP_LZ77/build/

for f in /scratch/ktruong/Workspace/experiments/078_EXPE_comp_measure_z_2/data/batches_concat/*.fa; do ./parse_and_compute.sh $f; done > /scratch/ktruong/Workspace/experiments/078_EXPE_comp_measure_z_2/results/number_of_phrases.txt
```
